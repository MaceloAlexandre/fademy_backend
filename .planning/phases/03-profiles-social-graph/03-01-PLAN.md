    follow.controller.js exports: follow, unfollow, acceptFollow, rejectFollow.

    follow(req, res):
      - followerId = req.user.id, followingId = req.params.id
      - Guard: followerId === followingId → 400 { error: 'Cannot follow yourself' }
      - Guard: Block.findOne({ $or: [{blockerId:followerId,blockedId:followingId},{blockerId:followingId,blockedId:followerId}] }) → if exists → 403
      - User.findById(followingId) → if not found → 404
      - status = target.isPrivate ? 'pending' : 'accepted'
      - Follow.create({ followerId, followingId, status }) inside try/catch
      - On success: notificationService.create(status==='accepted'?'new_follower':'follow_request', followerId, followingId, follow._id, 'Follow') — do NOT await inside the outer try (notification failure must not fail the action)
      - Return 201 { status: follow.status, followId: follow._id }
      - On err.code === 11000 → 409 { error: 'Already following or request pending' }

    unfollow(req, res):
      - Find Follow where { followerId: req.user.id, followingId: req.params.id } and deleteOne
      - Return 200 { unfollowed: true } regardless of whether a record existed (idempotent)

    acceptFollow(req, res):
      - followId = req.params.id (the Follow document _id, NOT a userId)
      - Follow.findById(followId) → if not found → 404
      - IDOR check: if (follow.followingId.toString() !== req.user.id) → 403 { error: 'Forbidden' }
      - Guard: if (follow.status !== 'pending') → 400 { error: 'No pending request' }
      - follow.status = 'accepted'; follow.save()
      - notificationService.create('follow_accepted', follow.followingId, follow.followerId, follow._id, 'Follow')
      - Return 200 { status: 'accepted' }

    rejectFollow(req, res):
      - followId = req.params.id
      - Follow.findById(followId) → if not found → 404
      - IDOR check: follow.followingId.toString() !== req.user.id → 403
      - Follow.findByIdAndDelete(followId)
      - Return 200 { rejected: true }

    block.controller.js exports: block.

    block(req, res):
      - blockerId = req.user.id, blockedId = req.params.id
      - Guard: blockerId === blockedId → 400 { error: 'Cannot block yourself' }
      - Follow.deleteMany({ $or: [{followerId:blockerId,followingId:blockedId},{followerId:blockedId,followingId:blockerId}] })
      - Block.findOneAndUpdate({ blockerId, blockedId }, { blockerId, blockedId }, { upsert: true, new: true })
      - Return 200 { blocked: true }

|----------|-------------|
| Client → POST /api/users/:id/follow | Attacker passes arbitrary userId as :id to follow self or blocked user |
| Client → PATCH /api/follows/:id/accept | Attacker passes arbitrary Follow _id to accept someone else's follow request (IDOR) |
| Client → POST /api/users/:id/block | Attacker passes their own id as :id (self-block) |
